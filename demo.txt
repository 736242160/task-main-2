# ---- 塘定义: 名称 容量m3 鱼种 密度kg/m3 ----
pond P1 5000 草鱼 12        # 高密度塘(阈值: 溶氧>=6.0, 氨氮<=0.2)
pond P2 3000 鲫鱼 5         # 低密度塘(阈值: 溶氧>=4.0, 氨氮<=0.5)
# ---- 设备定义: 名称 类型 服务塘 ----
device O1 aerator P1,P2
device W1 exchanger P1
device F1 feeder P1
# ---- 投喂流 ----
feed P1 2000                # 限量=3%*60000=1800kg -> 超限报告
feed P2 50
# ---- 监测流: 溶氧/氨氮级联 ----
monitor P1 5.2 26 0.30      # 溶氧低->级联开增氧; 氨氮高->级联开换水
monitor P1 5.5 26 0.20      # 增氧/换水生效, 水质逐步恢复
monitor P1 6.9 26 0.08      # 双指标恢复 -> 级联关闭设备
# ---- 故障流: 服务塘多参数级联告警 ----
fault O1                    # P1/P2 均报设备故障
monitor P1 5.0 26 0.10      # 溶氧低 + 增氧设备不可用(级联升级)
feed P1 100                 # 正常(投喂机未故障)
fault F1
feed P1 100                 # 投喂设备故障 -> 投喂失败
# ---- 修复后级联解除 ----
repair O1                   # 故障告警解除, 自动恢复增氧
monitor P1 6.8 26 0.08      # 溶氧恢复 -> 级联关闭
repair F1
feed P1 100                 # 恢复可投喂
# ---- 高/低密度阈值差异 ----
monitor P2 4.5 25 0.40      # 低密度阈值下正常(在 P1 会双告警)
monitor P2 3.5 25 0.60      # 双超 -> 告警并级联
# ---- 矛盾与未知引用 ----
monitor P9 5.0 25 0.10      # 不存在的塘
monitor P1 12.0 26 0.05     # 溶氧超饱和 -> 矛盾
monitor P2 1.5 25 0.05      # 溶氧极低但氨氮极低 -> 矛盾
status
